flowchart LR
  A[Customer browses] --> B[Adds to cart]
  B --> C{Coupon valid?}
  C -- Yes --> D[Apply discount]
  C -- No --> E[Pay full price]
  D --> F[Pay]
  E --> F
  F --> G{Payment ok?}
  G -- No --> B
  G -- Yes --> H[Restaurant accepts]
  H --> I[Rider assigned]
  I --> J[Out for delivery]
  J --> K[Delivered and rated]

    int order_id PK
    int customer_id FK
    int restaurant_id FK
    decimal amount
    int delivery_min
    string status

    int customer_id PK
    string name
    date joined_on
